# ১০. Understanding Pointers in Go

ধরো তোমার বন্ধুকে একটা বই ধার দিতে চাও। দুইভাবে দিতে পারো — বইয়ের একটা ফটোকপি করে দিতে পারো, অথবা আসল বইটাই দিতে পারো, শুধু ঠিকানা বলে দিয়ে "এই তাকে রাখা আছে, গিয়ে নিয়ে এসো।" প্রথমটা হলো **pass by value**, দ্বিতীয়টা হলো **pointer**-এর ধারণা — জিনিসটা না দিয়ে, জিনিসটা কোথায় আছে সেই ঠিকানা দেওয়া।

Go-তে প্রতিটা ভ্যারিয়েবল কম্পিউটারের মেমোরিতে একটা নির্দিষ্ট জায়গা দখল করে থাকে, আর সেই জায়গার একটা ঠিকানা (address) থাকে। সাধারণত আমরা ভ্যারিয়েবলের নাম দিয়ে তার ভ্যালু নিয়ে কাজ করি, কিন্তু মাঝে মাঝে আমাদের দরকার হয় সেই ভ্যালুর ঠিকানাটা নিয়ে কাজ করার — আর এখানেই পয়েন্টার আসে।

```go
package main

import "fmt"

func main() {
    age := 25
    ptr := &age // & মানে "এর ঠিকানা দাও"

    fmt.Println("ভ্যালু:", age)
    fmt.Println("ঠিকানা:", ptr)
    fmt.Println("ঠিকানা থেকে ভ্যালু:", *ptr) // * মানে "এই ঠিকানায় যা আছে তা দাও"

    *ptr = 30 // ঠিকানার মাধ্যমে ভ্যালু পরিবর্তন
    fmt.Println("নতুন ভ্যালু:", age)
}
```

এখানে দুইটা চিহ্ন গুরুত্বপূর্ণ — `&` (ampersand) দিয়ে আমরা কোনো ভ্যারিয়েবলের ঠিকানা বের করি, আর `*` (asterisk) দিয়ে সেই ঠিকানায় গিয়ে আসল ভ্যালুটা পড়ি বা বদলাই। একে বলে **dereferencing**।

এখন প্রশ্ন হলো, C ভাষায় তো পয়েন্টার নিয়ে অনেক ঝামেলা — pointer arithmetic (যেমন `ptr + 1` লিখে পরের মেমোরি ব্লকে চলে যাওয়া) করতে গিয়ে প্রোগ্রামাররা প্রায়ই ভুল মেমোরিতে ঢুকে পড়ে, প্রোগ্রাম ক্র্যাশ করে। Go-এর ডিজাইনাররা ইচ্ছাকৃতভাবে এই সুবিধাটা বাদ দিয়েছেন। Go-তে তুমি একটা পয়েন্টারকে বাড়াতে-কমাতে পারবে না, শুধু তার ভ্যালু পড়তে আর বদলাতে পারবে। এটা অনেকটা গাড়ি চালানোর মতো — স্টিয়ারিং হুইল দেওয়া হয়েছে দিক পরিবর্তনের জন্য, কিন্তু ইঞ্জিনের ভেতরে হাত ঢুকিয়ে সরাসরি নাড়াচাড়া করার অনুমতি নেই। নিরাপত্তা আগে।

এবার আসি ফাংশন কলে পয়েন্টারের আসল ব্যবহারে। Go-তে ডিফল্টভাবে ফাংশনে আর্গুমেন্ট পাঠালে সেটার একটা **কপি** তৈরি হয় — একে বলে **pass by value**।

```go
func doubleValue(n int) {
    n = n * 2 // এটা শুধু কপি বদলাচ্ছে, আসল ভ্যারিয়েবল অক্ষত
}

func doubleValuePointer(n *int) {
    *n = *n * 2 // এটা আসল ভ্যারিয়েবলকেই বদলে দিচ্ছে
}

func main() {
    x := 10
    doubleValue(x)
    fmt.Println(x) // এখনও 10

    doubleValuePointer(&x)
    fmt.Println(x) // এখন 20
}
```

লক্ষ করো, দ্বিতীয় ফাংশনে আমরা `x`-এর কপি না পাঠিয়ে তার ঠিকানা পাঠিয়েছি, তাই ফাংশনের ভেতরের পরিবর্তন বাইরের ভ্যারিয়েবলেও প্রতিফলিত হয়েছে। একে বলা যায় **pass by reference**-এর অনুকরণ, যদিও Go টেকনিক্যালি সবসময় pass by value — শুধু কখনো ভ্যালুটা নিজে, কখনো ভ্যালুর ঠিকানা।

এই একই যুক্তি কাজে লাগে struct-এর সাথে। Lesson ০৬-এ আমরা struct নিয়ে কথা বলেছিলাম — struct একটা বড় জিনিস হতে পারে, অনেকগুলো ফিল্ড নিয়ে গঠিত। যদি ফাংশনে struct-টা সরাসরি পাঠাও, পুরো জিনিসটার কপি তৈরি হবে, যা মেমোরির জন্য অপচয়। তাই বড় struct নিয়ে কাজ করার সময় সাধারণত **pointer receiver** ব্যবহার করা হয়:

```go
type Account struct {
    Balance float64
}

func (a *Account) Deposit(amount float64) {
    a.Balance += amount // মূল অ্যাকাউন্টের ব্যালেন্স আপডেট হচ্ছে
}
```

যদি এখানে `(a Account)` (pointer ছাড়া) লিখতাম, `Deposit` মেথডটা শুধু একটা কপির উপর কাজ করতো, আর আসল অ্যাকাউন্টের ব্যালেন্স কখনোই বদলাতো না — টাকা জমা পড়েও যেন পড়লো না! তাই যখনই কোনো মেথডের ভেতরে struct-এর ডেটা বদলাতে হয়, pointer receiver ব্যবহার করাই নিয়ম।

পরের লেসনে আমরা দেখবো Go-তে ডেটার তালিকা সামলানোর দুইটা ধরন — array আর slice — এবং কেন slice-ই বাস্তব প্রজেক্টে বেশি ব্যবহৃত হয়।
